JavaScript:

let indiceEdicao = null;
let atletas = [];

const formCadastro = document.getElementById('form-cadastro');
const tabelaAlunos = document.getElementById('tabela-alunos');
const inputNome = document.getElementById('nome');
const inputTurma = document.getElementById('turma');
const selectModalidade = document.getElementById('modalidade');
const btnSubmit = document.querySelector('.btn-submit');
const cardsMvpContainer = document.getElementById('cards-mvp');

// Função para atualizar a tabela de atletas cadastrados
function renderizarTabela() {
    tabelaAlunos.innerHTML = '';
    
    atletas.forEach((atleta, index) => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${atleta.nome}</td>
            <td>${atleta.turma}</td>
            <td>${atleta.modalidade}</td>
            <td>
                <button onclick="editarAtleta(${index})" style="background-color: #f39c12; color: white; border: none; padding: 5px 10px; border-radius: 4px; cursor: pointer; margin-right: 5px;">Editar</button>
                <button onclick="excluirAtleta(${index})" style="background-color: #e74c3c; color: white; border: none; padding: 5px 10px; border-radius: 4px; cursor: pointer;">Excluir</button>
            </td>
        `;
        tabelaAlunos.appendChild(tr);
    });
}

// Função para atualizar os cards de votação do MVP na tela
function renderizarMVP() {
    cardsMvpContainer.innerHTML = '';

    if (atletas.length === 0) {
        cardsMvpContainer.innerHTML = `
            <p class="section-subtitle">
                Nenhum atleta cadastrado para votação no momento.
            </p>
        `;
        return;
    }

    atletas.forEach((atleta, index) => {
        const card = document.createElement('div');

        card.classList.add('mvp-card');

        card.innerHTML = `
            <div class="card-content">

                <h3>${atleta.nome}</h3>

                <p>
                    <strong>Turma:</strong>
                    ${atleta.turma}
                </p>

                <p>
                    <strong>Modalidade:</strong>
                    ${atleta.modalidade}
                </p>

                <button 
                    onclick="votarMVP(${index})"
                    class="btn-votar-mvp"
                >
                    ⭐ Votar no MVP
                </button>

                <span id="votos-${index}" class="votos-mvp">
                    Votos: ${atleta.votos || 0}
                </span>

            </div>
        `;

        cardsMvpContainer.appendChild(card);
    });
}


// Evento do formulário de cadastro ou salvamento de edição
formCadastro.addEventListener('submit', function(e) {
    e.preventDefault();

    const nome = inputNome.value;
    const turma = inputTurma.value;
    const modalidade = selectModalidade.value;

    if (indiceEdicao !== null) {
        // Mantém a quantidade de votos anterior ao editar o atleta
        const votosAtuais = atletas[indiceEdicao].votos || 0;
        atletas[indiceEdicao] = { nome, turma, modalidade, votos: votosAtuais };
        indiceEdicao = null;
        btnSubmit.textContent = "Finalizar Inscrição";
    } else {
        atletas.push({ nome, turma, modalidade, votos: 0 });
    }

    formCadastro.reset();
    renderizarTabela();
    renderizarMVP();
});

// Ação de carregar dados para edição
window.editarAtleta = function(index) {
    const atleta = atletas[index];
    
    inputNome.value = atleta.nome;
    inputTurma.value = atleta.turma;
    selectModalidade.value = atleta.modalidade; // Carrega a modalidade correta no select

    indiceEdicao = index;
    btnSubmit.textContent = "Salvar Alterações";

    document.getElementById('cadastro').scrollIntoView({ behavior: 'smooth' });
};

// Ação de excluir atleta
window.excluirAtleta = function(index) {
    if (confirm("Tem certeza que deseja excluir este atleta?")) {
        atletas.splice(index, 1);
        if (indiceEdicao === index) {
            formCadastro.reset();
            indiceEdicao = null;
            btnSubmit.textContent = "Finalizar Inscrição";
        }
        renderizarTabela();
        renderizarMVP();
    }
};

// Ação de computar voto no MVP
window.votarMVP = function(index) {
    if (!atletas[index].votos) {
        atletas[index].votos = 0;
    }
    atletas[index].votos++;

    const spanVotos = document.getElementById(`votos-${index}`);
    if (spanVotos) {
        spanVotos.textContent = `Votos: ${atletas[index].votos}`;
    }
};

// Inicializa a tela vazia
renderizarMVP();
