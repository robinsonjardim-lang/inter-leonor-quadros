Código do Google Apps Script (Code.gs) Atualizado

Copie e atualize o código abaixo no seu editor do Google Apps Script vinculado à Planilha Google (Aba Atletas):

// ========================================================
// GOOGLE APPS SCRIPT - INTERCLASSE LEONOR QUADROS 2026
// Cole este código no editor (Extensões > Apps Script)
// ========================================================

function doGet(e) {
  var sheet = getOrCreateSheet();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  
  // Pula a linha do cabeçalho (Linha 1)
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) {
      // Força a leitura limpa do ID como String
      var rawId = String(data[i][0]).replace(/^'/, '').trim();
      rows.push({
        id: rawId,
        nome: String(data[i][1] || ""),
        turma: String(data[i][2] || ""),
        modalidade: String(data[i][3] || ""),
        votos: Number(data[i][4]) || 0,
        detalhes: String(data[i][5] || "")
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = getOrCreateSheet();
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;

    if (action === "cadastrar") {
      // Salva o ID precedido por aspa simples para forçar texto no Google Sheets
      var athleteId = contents.id ? String(contents.id) : Date.now().toString();
      var formattedId = "'" + athleteId;
      
      sheet.appendRow([
        formattedId,
        contents.nome || "",
        contents.turma || "",
        contents.modalidade || "",
        0, // Votos iniciais (Coluna E - 5)
        contents.detalhes || "" // Descrição/Detalhes do Projeto (Coluna F - 6)
      ]);
      return responseJSON({ status: "success", message: "Inscrição efetuada com sucesso!", id: athleteId });
    }

    if (action === "votar") {
      var targetId = String(contents.id).trim();
      var data = sheet.getDataRange().getValues();

      for (var i = 1; i < data.length; i++) {
        var cellId = String(data[i][0]).replace(/^'/, '').trim();
        if (cellId === targetId) {
          var currentVotes = Number(data[i][4]) || 0;
          var newVotes = currentVotes + 1;
          sheet.getRange(i + 1, 5).setValue(newVotes); // Atualiza Coluna E (Votos)
          return responseJSON({ status: "success", id: targetId, newVotes: newVotes });
        }
      }
      return responseJSON({ status: "error", message: "Atleta com o ID não foi localizado na planilha." });
    }

    return responseJSON({ status: "error", message: "Ação não reconhecida." });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Atletas");
  if (!sheet) {
    sheet = ss.insertSheet("Atletas");
    // Estrutura das colunas: A=ID, B=Nome, C=Turma, D=Modalidade, E=Votos, F=Detalhes
    sheet.appendRow(["ID", "Nome", "Turma", "Modalidade", "Votos", "Detalhes / Descrição do Projeto"]);
  }
  return sheet;
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

Passos cruciais após colar o código:

  1. Clique em Salvar (ícone de disquete).
  2. Clique no botão azul Implantar > Gerenciar implantações.
  3. Clique no ícone de Lápis (Editar) na implantação existente.
  4. Em Versão, selecione Nova Versão (obrigatório para aplicar alterações!).
  5. Verifique se Quem pode acessar está como Qualquer pessoa (Anyone).
  6. Clique em Implantar.
